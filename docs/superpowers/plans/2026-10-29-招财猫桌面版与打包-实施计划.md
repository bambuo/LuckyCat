# 招财猫桌面版与打包 · 实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 为 LuckyCat 增加 Avalonia 桌面外壳（同进程内嵌 Kestrel、无边框标题栏、托盘常驻），产出 Windows/macOS/Linux 三平台 Velopack 安装包与 GitHub Releases 自动更新。

**Architecture:** 现有 ASP.NET Core 管道抽成 `LuckyCatHost.Build(args, options)`，CLI 与桌面共用；桌面工程以 `WinExe` 启动 Avalonia，同进程 `StartAsync` 监听 `127.0.0.1:0`，`NativeWebView` 加载同源前端；标题栏由 Avalonia 自绘并通过消息桥与网页路由双向同步；Velopack 负责安装与更新。

**Tech Stack:** .NET 10、ASP.NET Core Minimal API、SQLite/EF Core、Vue 3 + Vite、Avalonia 12.1.4、Avalonia.Controls.WebView 12.1.1、Velopack 1.2.161、xunit 2.9.3、bun。

**Spec:** `docs/superpowers/specs/2026-10-29-招财猫桌面版与打包-设计.md`

## Global Constraints

- 目标框架 `net10.0`；本机 SDK 在 `~/.dotnet`，每条命令前执行 `export DOTNET_ROOT="$HOME/.dotnet"; export PATH="$HOME/.dotnet:$PATH"`。
- 版本唯一来源为 `Directory.Build.props` 的 `<Version>`，初始 `0.1.0`；打包脚本从该文件读取。
- 桌面模式只监听 `127.0.0.1`，端口由系统分配（`http://127.0.0.1:0`）；CLI 模式固定 `http://localhost:5210`。
- 数据目录只使用 `LocalApplicationData/LuckyCat`，不读取、不复制仓库根的 `luckycat.db`。
- 不使用透明窗口与自绘圆角；圆角、阴影、贴边分屏交给系统。
- 提示类 UI（更新、错误、首次关窗）必须是独立 `Window`，不得叠加在 `NativeWebView` 上。
- 更新只自动下载，重启必须由用户确认；`CycleStartupRecovery` 的恢复语义不得改动。
- 不新增 API 鉴权、不改现有端点契约、不改前端业务页面。
- 每个任务最后的提交步骤需用户明确授权后执行；未获授权时停在「验证通过」并汇报。

// Hosting/StaticFrontend.cs
using Microsoft.AspNetCore.Builder;
using Microsoft.AspNetCore.Http;
using Microsoft.Extensions.FileProviders;
using LuckyCat.Contracts;

namespace LuckyCat.Hosting;

public static class StaticFrontend
{
    public static void UseLuckyCatFrontend(WebApplication app, string webRoot)
    {
        if (!Directory.Exists(webRoot)) return;

        var provider = new PhysicalFileProvider(webRoot);
        app.UseDefaultFiles(new DefaultFilesOptions { FileProvider = provider });
        app.UseStaticFiles(new StaticFileOptions { FileProvider = provider });
    }

    public static void MapLuckyCatFrontend(WebApplication app, string webRoot)
    {
        if (!Directory.Exists(webRoot)) return;

        var provider = new PhysicalFileProvider(webRoot);

        // 先注册 API 兜底：未匹配的 /api 路径返回 JSON 404，不被 index.html 吞掉。
        app.MapFallback("/api/{**rest}", () =>
            Results.Json(ApiResponse<object>.Fail("not_found", "接口不存在"), ApiJson.Options, statusCode: 404));

        app.MapFallbackToFile("index.html", new StaticFileOptions { FileProvider = provider });
    }
}
```

// web/src/desktop.ts
const DESKTOP_KEY = 'luckycat:desktop'

export function parseDesktopFlag(search: string, stored: string | null): boolean {
  const requested = new URLSearchParams(search).get('desktop') === '1'
  return requested || stored === '1'
}

export function isDesktopMode(): boolean {
  if (typeof window === 'undefined') return false
  const flag = parseDesktopFlag(window.location.search, window.sessionStorage.getItem(DESKTOP_KEY))
  if (flag) window.sessionStorage.setItem(DESKTOP_KEY, '1')
  return flag
}

export function routeMessage(path: string): string {
  return JSON.stringify({ type: 'route', path })
}

export function notifyRoute(path: string): void {
  window.invokeCSharpAction?.(routeMessage(path))
}

export function installDesktopNavigation(push: (path: string) => void): void {
  window.__lcNavigate = (path: string) => push(path)
}

declare global {
  interface Window {
    invokeCSharpAction?: (data: string) => void
    __lcNavigate?: (path: string) => void
  }
}
```

using System.Globalization;

namespace LuckyCat.Desktop.Core;

public static class LogRetention
{
    private const string Prefix = "app-";
    private const string Extension = ".log";
    private const string Format = "yyyyMMdd";

    public static int DeleteOlderThan(string logDirectory, DateTimeOffset now, int keepDays)
    {
        if (!Directory.Exists(logDirectory)) return 0;

        var cutoff = DateOnly.FromDateTime(now.UtcDateTime).AddDays(-keepDays);
        var deleted = 0;

        foreach (var file in Directory.EnumerateFiles(logDirectory, "app-*.log"))
        {
            var name = Path.GetFileNameWithoutExtension(file);
            if (!name.StartsWith(Prefix, StringComparison.Ordinal)) continue;
            if (!DateOnly.TryParseExact(name[Prefix.Length..], Format, CultureInfo.InvariantCulture,
                    DateTimeStyles.None, out var day)) continue;
            if (day >= cutoff) continue;

            File.Delete(file);
            deleted++;
        }

        return deleted;
    }
}

namespace LuckyCat.Desktop.Core;

public static class ConsoleRedirect
{
    public static IDisposable ToFile(string logDirectory, DateTimeOffset now)
    {
        Directory.CreateDirectory(logDirectory);
        var path = Path.Combine(logDirectory, $"app-{now:yyyyMMdd}.log");
        var writer = new StreamWriter(
            new FileStream(path, FileMode.Append, FileAccess.Write, FileShare.Read),
            new UTF8Encoding(encoderShouldEmitUTF8Identifier: false))
        {
            AutoFlush = true,
        };

        var originalOut = Console.Out;
        var originalError = Console.Error;
        var sync = TextWriter.Synchronized(writer);
        Console.SetOut(sync);
        Console.SetError(sync);

        return new Restore(() =>
        {
            Console.SetOut(originalOut);
            Console.SetError(originalError);
            writer.Dispose();
        });
    }

    private sealed class Restore(Action action) : IDisposable
    {
        private bool _disposed;

        public void Dispose()
        {
            if (_disposed) return;
            _disposed = true;
            action();
        }
    }
}

public sealed class DesktopHost : IAsyncDisposable
{
    private readonly WebApplication _app;

    private DesktopHost(WebApplication app, Uri address)
    {
        _app = app;
        Address = address;
    }

    public Uri Address { get; }

    public static async Task<DesktopHost> StartAsync(string dataDirectory, CancellationToken ct = default)
    {
        Directory.CreateDirectory(dataDirectory);
        Directory.CreateDirectory(DesktopPaths.LogDirectory(dataDirectory));
        LogRetention.DeleteOlderThan(DesktopPaths.LogDirectory(dataDirectory), DateTimeOffset.Now, 7);

        var app = LuckyCatHost.Build([], LuckyCatHostOptions.Desktop(dataDirectory));

        // 二次启动唤起：仅桌面模式注册，只对回环来源生效。
        app.MapPost("/internal/activation", (HttpContext context) =>
            context.Connection.RemoteIpAddress is { } ip && IPAddress.IsLoopback(ip)
                ? Results.NoContent()
                : Results.NotFound());

        await app.StartAsync(ct);

        var server = app.Services.GetRequiredService<IServer>();
        var addresses = server.Features.Get<IServerAddressesFeature>()?.Addresses;
        var address = new Uri(addresses?.FirstOrDefault()
            ?? throw new InvalidOperationException("Kestrel 未报告监听地址"));

        InstanceFile.Write(DesktopPaths.InstanceInfoPath(dataDirectory),
            new InstanceInfo(address.ToString(), Environment.ProcessId));

        return new DesktopHost(app, address);
    }

    public async ValueTask DisposeAsync()
    {
        await _app.StopAsync(TimeSpan.FromSeconds(5));
        await _app.DisposeAsync();
    }

    {
        if (string.IsNullOrWhiteSpace(e.Body)) return;

        {
            using var document = JsonDocument.Parse(e.Body);

            if (root.TryGetProperty("type", out var type) && type.GetString() == "route"
                && root.TryGetProperty("path", out var path) && path.GetString() is { } value)
            {
                RouteChanged?.Invoke(value);
            }
        }
        catch (JsonException)
        {
            // 未知负载忽略：消息桥只服务标题栏选中态。
        }
    }

using System.Diagnostics;

namespace LuckyCat.Desktop;

public static class StatusNotifierProbe
{
    public static bool Available()
    {
        try
        {
            var startInfo = new ProcessStartInfo("dbus-send")
            {
                RedirectStandardOutput = true,
                UseShellExecute = false,
            };
            startInfo.ArgumentList.Add("--session");
            startInfo.ArgumentList.Add("--dest=org.freedesktop.DBus");
            startInfo.ArgumentList.Add("--print-reply");
            startInfo.ArgumentList.Add("--type=method_call");
            startInfo.ArgumentList.Add("/org/freedesktop/DBus");
            startInfo.ArgumentList.Add("org.freedesktop.DBus.NameHasOwner");
            startInfo.ArgumentList.Add("string:org.kde.StatusNotifierWatcher");

            using var process = Process.Start(startInfo);
            if (process is null) return false;
            var output = process.StandardOutput.ReadToEnd();
            process.WaitForExit(2000);
            return output.Contains("boolean true", StringComparison.Ordinal);
        }
        catch (Exception ex) when (ex is System.ComponentModel.Win32Exception or InvalidOperationException)
        {
            return false;
        }
    }
}

```csharp
// desktop/LuckyCat.Desktop/Updater.cs
using Avalonia.Controls;
using Avalonia.Threading;
using Velopack;
using Velopack.Sources;

namespace LuckyCat.Desktop;

public static class Updater
{
    private static readonly UpdateManager Manager =
        new(new GithubSource("https://github.com/bambuo/LuckyCat", accessToken: null, prerelease: false));

    public static async Task CheckNowAsync(Window owner)
    {
        try
        {
            if (!Manager.IsInstalled) return;

            var update = await Manager.CheckForUpdatesAsync();
            if (update is null) return;

            await Manager.DownloadUpdatesAsync(update);
            await Dispatcher.UIThread.InvokeAsync(() => PromptRestart(owner, update));
        }
        catch (Exception)
        {
            // 更新失败不打断交易：只落日志。
            Console.WriteLine("[Updater] 检查或下载更新失败");
        }
    }

    public static async Task StartPeriodic(Window owner, CancellationToken ct)
    {
        await Task.Delay(TimeSpan.FromSeconds(60), ct);
        while (!ct.IsCancellationRequested)
        {
            await CheckNowAsync(owner);
            await Task.Delay(TimeSpan.FromHours(6), ct);
        }
    }

    private static void PromptRestart(Window owner, UpdateInfo update)
    {
        var window = new NoticeWindow { Title = "发现新版本" };
        window.Heading.Text = $"新版本 {update.TargetFullRelease.Version} 已下载";
        window.Body.Text =
            "重启后生效。若交易员正在运行，重启会中断当前周期；重启后运行中的交易员会自动恢复调度。";
        window.PrimaryAction.Content = "立即重启";
        window.PrimaryAction.IsVisible = true;
        window.RestartRequested += () =>
        {
            Manager.ApplyUpdatesAndRestart(update.TargetFullRelease);
        };
        window.Show(owner);
    }
}
```

`NoticeWindow` 的 `PrimaryAction` 与 `RestartRequested` 已在 Task 6 提供，本任务只使用，不再修改该文件。

// scripts/make-ico.mjs —— 把多个 PNG 打包成 ICO（PNG 压缩的 ICO 合法且为 Win10+ 支持）
import { readFileSync, writeFileSync } from 'node:fs'
import { join } from 'node:path'

const [workDir, outFile, ...sizes] = process.argv.slice(2)
const images = sizes.map((size) => ({
  size: Number(size),
  data: readFileSync(join(workDir, `app-${size}.png`)),
}))

const header = Buffer.alloc(6)
header.writeUInt16LE(0, 0)
header.writeUInt16LE(1, 2)
header.writeUInt16LE(images.length, 4)

const entries = Buffer.alloc(16 * images.length)
let offset = header.length + entries.length

images.forEach((image, index) => {
  const at = index * 16
  entries.writeUInt8(image.size >= 256 ? 0 : image.size, at)
  entries.writeUInt8(image.size >= 256 ? 0 : image.size, at + 1)
  entries.writeUInt8(0, at + 2)
  entries.writeUInt8(0, at + 3)
  entries.writeUInt16LE(1, at + 4)
  entries.writeUInt16LE(32, at + 6)
  entries.writeUInt32LE(image.data.length, at + 8)
  entries.writeUInt32LE(offset, at + 12)
  offset += image.data.length
})

writeFileSync(outFile, Buffer.concat([header, entries, ...images.map((image) => image.data)]))
console.log(`ico written: ${outFile}`)
